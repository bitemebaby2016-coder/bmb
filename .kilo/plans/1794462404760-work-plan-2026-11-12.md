เพิ่ม `localStorage` fallback สำหรับ `?cart=` data:

```typescript
// R96 Cart Continuity Fix (2026-11-12): localStorage fallback สำหรับ Facebook IAB
const CART_STORAGE_KEY = 'bmb:cart-from-link'

// เมื่อเจอ ?cart= ใน URL → เก็บลง localStorage ด้วย
if (raw) {
  try {
    const decoded = decodeCartParam(raw)
    if (decoded && decoded.length > 0) {
      sessionStorage.setItem(CART_STORAGE_KEY, JSON.stringify({ items: decoded, at: Date.now() }))
    }
  } catch { /* ignore */ }
}

// ถ้า URL param หาย (Facebook IAB strip) → อ่านจาก localStorage
const restoredItems = () => {
  // Try URL param first
  if (raw) {
    const items = decodeCartParam(raw)
    if (items && items.length > 0) return items
  }
  // Fallback to localStorage
  try {
    const stored = sessionStorage.getItem(CART_STORAGE_KEY)
    if (stored) {
      const data = JSON.parse(stored)
      if (data.items && data.items.length > 0) return data.items
    }
  } catch { /* ignore */ }
  return null
}
```

### Step 6: Cache Invalidation on Product Update

**กลยุทธ์:** ใช้ session-level cache busting แทน persistent cache

**ไฟล์:** สร้าง `src/lib/productCache.ts`

```typescript
// Simple session-based cache with explicit invalidate
let cachedProducts: Product[] | null = null
let cacheTimestamp: number = 0
const CACHE_TTL_MS = 7 * 60 * 1000 // 7 minutes (same as dbContext)

export async function getCachedProducts(): Promise<Product[]> {
  const now = Date.now()
  if (cachedProducts && now - cacheTimestamp < CACHE_TTL_MS) {
    return cachedProducts
  }
  cachedProducts = await fetchProducts() // real API call
  cacheTimestamp = now
  return cachedProducts
}

export function invalidateProductCache(): void {
  cachedProducts = null
  cacheTimestamp = 0
}
```

ใช้ใน `chatWithAI()` → เรียก `getCachedProducts()` แทน `getDbContextPrompt()` สำหรับ product data
ใช้ใน admin pages → หลัง save promotion/product → เรียก `invalidateProductCache()`

#### Step 2: Add Order Memory RPCs

**ไฟล์:** `supabase/functions/_shared/orderMemory.ts`

```typescript
// R97: Order memory RPCs for AI worker

export interface OrderSummary {
  order_id: string
  items: Array<{ name: string; quantity: number; price: number }>
  total: number
  status: 'pending' | 'confirmed' | 'processing' | 'delivered' | 'cancelled'
  created_at: string
}

/** Get order history for a customer (last 10 orders) */
export async function getCustomerOrders(
  sbUrl: string,
  serviceKey: string,
  customerId: string,
): Promise<OrderSummary[]> {
  const res = await fetch(
    `${sbUrl}/rest/v1/orders?select=order_id,items,total,status,created_at&customer_id=eq.${encodeURIComponent(customerId)}&order=created_at.desc&limit=10`,
    {
      headers: { apikey: serviceKey, Authorization: `Bearer ${serviceKey}` },
      signal: AbortSignal.timeout(5000),
    },
  )
  if (!res.ok) return []
  const data = await res.json()
  return (data || []).map((o: any) => ({
    order_id: o.order_id,
    items: o.items || [],
    total: o.total,
    status: o.status,
    created_at: o.created_at,
  }))
}

/** Cancel an order (soft delete — mark as cancelled) */
export async function cancelOrder(
  sbUrl: string,
  serviceKey: string,
  orderId: string,
  reason?: string,
): Promise<{ ok: boolean; error?: string }> {
  try {
    const res = await fetch(`${sbUrl}/rest/v1/orders?order_id=eq.${encodeURIComponent(orderId)}`, {
      method: 'PATCH',
      headers: {
        apikey: serviceKey,
        Authorization: `Bearer ${serviceKey}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        status: 'cancelled',
        cancel_reason: reason || 'cancelled_by_customer',
        cancelled_at: new Date().toISOString(),
      }),
    })
    if (!res.ok) return { ok: false, error: `HTTP ${res.status}` }
    return { ok: true }
  } catch (e) {
    return { ok: false, error: String(e) }
  }
}

/** Update order items (add/remove/modify) */
export async function modifyOrder(
  sbUrl: string,
  serviceKey: string,
  orderId: string,
  newItems: Array<{ name: string; quantity: number; note?: string }>,
): Promise<{ ok: boolean; error?: string; newTotal?: number }> {
  try {
    // Calculate new total from items
    const { getProducts } = await import('../_shared/shopContext.ts')
    const products = await getProducts({ sbUrl, serviceKey })
    const productMap = new Map(products.map((p: any) => [p.name, p]))
    
    let newTotal = 0
    const updatedItems = newItems.map((item) => {
      const product = productMap.get(item.name)
      if (!product) return null
      const price = product.price * item.quantity
      newTotal += price
      return {
        product_id: product.id,
        name: item.name,
        quantity: item.quantity,
        note: item.note,
        price,
      }
    }).filter(Boolean)
    
    const res = await fetch(`${sbUrl}/rest/v1/orders?order_id=eq.${encodeURIComponent(orderId)}`, {
      method: 'PATCH',
      headers: {
        apikey: serviceKey,
        Authorization: `Bearer ${serviceKey}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        items: updatedItems,
        total: newTotal,
        status: 'pending', // Reset to pending for modification
      }),
    })
    if (!res.ok) return { ok: false, error: `HTTP ${res.status}` }
    return { ok: true, newTotal }
  } catch (e) {
    return { ok: false, error: String(e) }
  }
}
```
